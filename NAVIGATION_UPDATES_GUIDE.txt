#!/usr/bin/env python3
import os
import glob

new_footer_links = '''    <a href="faq.html" style="color:#667eea;margin:0 15px;text-decoration:none">FAQ & Help</a>
    <a href="blog.html" style="color:#667eea;margin:0 15px;text-decoration:none">Blog & Resources</a>
    <a href="alumni.html" style="color:#667eea;margin:0 15px;text-decoration:none">Our Alumni</a>
    <a href="refund-policy.html" style="color:#667eea;margin:0 15px;text-decoration:none">Refund Policy</a>'''

# Skip already updated files
skip_files = ['faq.html', 'blog.html', 'alumni.html', 'refund-policy.html', '404.html']

for html_file in glob.glob('*.html'):
    if html_file in skip_files:
        continue
    
    with open(html_file, 'r') as f:
        content = f.read()
    
    # Find the footer and add links before closing </footer>
    if '<a href="contact.html"' in content:
        # Add after existing footer links
        old_pattern = '    <a href="contact.html"'
        new_pattern = new_footer_links + '\n    <a href="contact.html"'
        content = content.replace(old_pattern, new_pattern, 1)
        
        with open(html_file, 'w') as f:
            f.write(content)
        print(f'✓ Updated {html_file}')
